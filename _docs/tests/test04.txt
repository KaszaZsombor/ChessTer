VÁRT EREDMÉNY: A beírt szöveg ha megfelel a program kiírja a koordinátákat,
a beírt lépésnek és a promotion piece számát, majd kirajzolja az új pozíciót


MAIN:
#include <stdio.h>
#include "board.h"
#include "types.h"
#include "move.h"
#include "ui.h"

int main(void){
    board main_brd = b_create();
    ui_drawboard(main_brd);
    b_setdefault(&main_brd);
    ui_drawboard(main_brd);

    int size = 6;
    char buffer[6];

    int result = ui_getusrin(&size, buffer);
    
    if(result == END)return 0;
    else if(result == ERROR)return 0;

    move mv;
    result = ui_parse_str(&mv, buffer, size);

    printf("%d%d\n", mv.from.row, mv.from.file);
    printf("%d%d\n", mv.to.row, mv.to.file);
    printf("%d\n", mv.promotion_piece);

    printf("%s\n", buffer);

    m_piece_to_p(&main_brd, mv.from, mv.to);
    ui_drawboard(main_brd);

    return 0;
}
